import React, { useState, useRef, useEffect, useCallback } from 'react';

interface TransformState {
  x: number;
  y: number;
  scale: number;
}

const INITIAL_STATE: TransformState = {
  x: 0, y: 0, scale: 1,
};

// 拖拽死区阈值 (像素)
const DRAG_THRESHOLD = 8;

// ==========================================
// 模块 1：ImageLayer (渲染层)
// ==========================================
interface ImageLayerProps {
  src: string | null;
  transform: TransformState;
}

const ImageLayer = React.forwardRef<HTMLImageElement, ImageLayerProps>(({ src, transform }, ref) => {
  if (!src) return null;
  return (
    <div 
      className="absolute inset-0 flex items-center justify-center pointer-events-none"
      style={{
        transform: `translate(${transform.x}px, ${transform.y}px)`,
        willChange: 'transform'
      }}
    >
      <img
        ref={ref}
        src={src}
        alt="Canvas Content"
        className="max-w-[90%] max-h-[90%] w-auto h-auto shadow-md pointer-events-none select-none"
        style={{
          transform: `scale(${transform.scale})`,
          transformOrigin: 'center center'
        }}
      />
    </div>
  );
});

// ==========================================
// 模块 2：InteractLayer (交互层)
// ==========================================
interface InteractLayerProps {
  transform: TransformState;
  onTransformChange: (updater: (prev: TransformState) => TransformState) => void;
  contentRef: React.RefObject<HTMLElement>;
  children: React.ReactNode;
}

const InteractLayer: React.FC<InteractLayerProps> = ({ 
  transform, 
  onTransformChange, 
  contentRef,
  children 
}) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const [isDragging, setIsDragging] = useState(false);
  const [hasExceededThreshold, setHasExceededThreshold] = useState(false);
  const [mouseStart, setMouseStart] = useState({ x: 0, y: 0 });
  const [initialTransform, setInitialTransform] = useState({ x: 0, y: 0 });

  const handleWheel = useCallback((e: React.WheelEvent) => {
    if (!containerRef.current || !contentRef.current) return;

    const contentRect = contentRef.current.getBoundingClientRect();
    const isInside = (
      e.clientX >= contentRect.left &&
      e.clientX <= contentRect.right &&
      e.clientY >= contentRect.top &&
      e.clientY <= contentRect.bottom
    );

    if (!isInside) return;

    e.preventDefault();
    const factor = Math.pow(1.1, (-e.deltaY) / 200);

    onTransformChange(prev => {
      const newScale = Math.min(Math.max(prev.scale * factor, 0.05), 50);
      const actualFactor = newScale / prev.scale;
      const rect = containerRef.current!.getBoundingClientRect();
      const centerX = rect.left + rect.width / 2;
      const centerY = rect.top + rect.height / 2;
      const mouseX = e.clientX - (centerX + prev.x);
      const mouseY = e.clientY - (centerY + prev.y);

      return {
        scale: newScale,
        x: prev.x - mouseX * (actualFactor - 1),
        y: prev.y - mouseY * (actualFactor - 1)
      };
    });
  }, [contentRef, onTransformChange]);

  const handleMouseDown = useCallback((e: React.MouseEvent) => {
    if (e.button !== 0) return; 
    e.preventDefault(); 
    
    setIsDragging(true);
    setHasExceededThreshold(false);
    
    // 记录鼠标按下的绝对坐标
    setMouseStart({ x: e.clientX, y: e.clientY });
    // 记录按下的那一刻图片的位移状态
    setInitialTransform({ x: transform.x, y: transform.y });
  }, [transform.x, transform.y]);

  useEffect(() => {
    if (!isDragging) return;

    const handleMouseMove = (e: MouseEvent) => {
      const dx = e.clientX - mouseStart.x;
      const dy = e.clientY - mouseStart.y;

      // 如果尚未超过死区，检查本次移动距离
      if (!hasExceededThreshold) {
        const distance = Math.sqrt(dx * dx + dy * dy);
        if (distance > DRAG_THRESHOLD) {
          setHasExceededThreshold(true);
        } else {
          return; // 依然在死区内，跳过更新
        }
      }

      // 超过死区后，根据初始位置和偏移量计算新位置
      onTransformChange(prev => ({
        ...prev,
        x: initialTransform.x + dx,
        y: initialTransform.y + dy
      }));
    };

    const handleMouseUp = () => {
      setIsDragging(false);
      setHasExceededThreshold(false);
    };

    window.addEventListener('mousemove', handleMouseMove);
    window.addEventListener('mouseup', handleMouseUp);
    return () => {
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('mouseup', handleMouseUp);
    };
  }, [isDragging, mouseStart, initialTransform, hasExceededThreshold, onTransformChange]);

  return (
    <div 
      ref={containerRef}
      className={`w-full h-full relative overflow-hidden bg-slate-100 touch-none select-none ${isDragging && hasExceededThreshold ? 'cursor-grabbing' : 'cursor-grab'}`}
      onMouseDown={handleMouseDown}
      onWheel={handleWheel}
    >
      {children}
    </div>
  );
};

// ==========================================
// 模块 3：ImageCanvas (核心逻辑组件)
// ==========================================
interface ImageCanvasProps {
  onLoadImage: () => Promise<string>;
}

const ImageCanvas: React.FC<ImageCanvasProps> = ({ onLoadImage }) => {
  const [transform, setTransform] = useState<TransformState>(INITIAL_STATE);
  const [imageSrc, setImageSrc] = useState<string | null>(null);
  const imgRef = useRef<HTMLImageElement>(null);

  useEffect(() => {
    let isMounted = true;
    
    if (typeof onLoadImage === 'function') {
      onLoadImage()
        .then(url => {
          if (isMounted) setImageSrc(url);
        })
        .catch(err => {
          console.error("Failed to load image in ImageCanvas:", err);
        });
    }

    return () => {
      isMounted = false;
    };
  }, [onLoadImage]);

  return (
    <div className="w-full h-full overflow-hidden">
      <InteractLayer 
        transform={transform} 
        onTransformChange={setTransform} 
        contentRef={imgRef}
      >
        <ImageLayer 
          ref={imgRef}
          src={imageSrc}
          transform={transform}
        />
      </InteractLayer>
    </div>
  );
};

// ==========================================
// 模块 4：App (入口组件)
// ==========================================
export default function App() {
  const handleLoadImage = async () => {
    return "https://images.unsplash.com/photo-1506744038136-46273834b3fb?auto=format&fit=crop&w=1200&q=80";
  };

  return (
    <div className="w-screen h-screen">
      <ImageCanvas onLoadImage={handleLoadImage} />
    </div>
  );
}
