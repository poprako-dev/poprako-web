import React from 'react';
import { Clock } from 'lucide-react';

/**
 * 职位解析函数
 */
function getRoleLabels(roles) {
  const roleMap = [
    { bit: 1, label: '图' },
    { bit: 2, label: '翻' },
    { bit: 4, label: '校' },
    { bit: 8, label: '嵌' },
    { bit: 16, label: '监' },
    { bit: 32, label: '传' },
  ];
  const labels = roleMap.filter(r => (roles & r.bit)).map(r => r.label);
  return labels.length > 0 ? labels : ['员'];
}

/**
 * 工作流状态判定工具函数
 */
export function uploadWorkflowStatus(chapter) {
  if (chapter.uploadedAt) return "completed";
  return "pending";
}

export function translateWorkflowStatus(chapter) {
  if (chapter.translatedAt) return "completed";
  if (chapter.transalatingAt) return "in_progress";
  return "pending";
}

export function proofreadWorkflowStatus(chapter) {
  if (chapter.proofreadAt) return "completed";
  if (chapter.proofreadingAt) return "in_progress";
  return "pending";
}

export function typesetWorkflowStatus(chapter) {
  if (chapter.typesetAt) return "completed";
  if (chapter.typesettingAt) return "in_progress";
  return "pending";
}

export function reviewWorkflowStatus(chapter) {
  if (chapter.reviewedAt) return "completed";
  return "pending";
}

export function publishWorkflowStatus(chapter) {
  if (chapter.publishedAt) return "completed";
  return "pending";
}

/**
 * ReviewerAssignmentCard 组件
 */
function ReviewerAssignmentCard({ assignmentInfo }) {
  const chapter = assignmentInfo?.chapter || {};
  const comic = chapter?.comic || {};
  const roles = assignmentInfo?.roles || 0;
  const pages = chapter.pageCount || 0;

  const formatDate = (ts) => {
    if (!ts) return '';
    const d = new Date(ts);
    return `${d.getFullYear()}/${d.getMonth() + 1}/${d.getDate()}`;
  };

  return (
    <div className="w-full h-[115px] bg-white border border-slate-100 rounded-lg flex flex-col transition-all hover:border-slate-200 hover:shadow-sm overflow-hidden group">
      <div className="flex h-full p-3 gap-4 items-center">
        {/* 左侧封面 */}
        <div className="w-18 h-full flex-shrink-0 bg-slate-50 rounded overflow-hidden border border-slate-100">
          <img 
            src={chapter.coverUrl || comic.coverUrl || 'https://via.placeholder.com/150x200?text=Cover'} 
            alt={comic.title}
            className="w-full h-full object-cover transition-transform duration-700 group-hover:scale-105"
          />
        </div>

        {/* 右侧详情 */}
        <div className="flex-1 flex flex-col justify-center gap-2 min-w-0">
          {/* 第一行：标题 + 话号 */}
          <div className="flex items-center gap-3">
            <h3 className="text-lg font-bold text-slate-700 truncate flex-1 leading-none">
              {comic.title || '未命名'}
            </h3>
            <span className="text-xs font-mono text-slate-400 bg-slate-50 border border-slate-100 px-2 py-0.5 rounded shadow-sm">
              #{chapter.chapterNo || '-'}
            </span>
          </div>

          {/* 第二行：时间 + 职位 + 页数 */}
          <div className="flex items-center gap-3 text-[11px] text-slate-400">
            <div className="flex items-center gap-1 flex-shrink-0">
              <Clock size={12} className="opacity-60" />
              <span className="font-mono tracking-tight">{formatDate(chapter.updatedAt)}</span>
            </div>
            <div className="h-3 w-[1px] bg-slate-100"></div>
            <div className="flex gap-1.5 font-medium text-slate-400/70">
              {getRoleLabels(roles).map((role, idx) => (
                <span key={idx}>{role}</span>
              ))}
            </div>
            <div className="h-3 w-[1px] bg-slate-100"></div>
            <div className="font-mono font-medium text-slate-400/70">
              {pages}P
            </div>
          </div>
          
          {/* 第三行：重新设计的 Workflow 进度条 (文字居中版) */}
          <div className="flex gap-1 w-full mt-1">
            <WorkflowTag label="图" status={uploadWorkflowStatus(chapter)} />
            <WorkflowTag label="翻" status={translateWorkflowStatus(chapter)} />
            <WorkflowTag label="校" status={proofreadWorkflowStatus(chapter)} />
            <WorkflowTag label="嵌" status={typesetWorkflowStatus(chapter)} />
            <WorkflowTag label="监" status={reviewWorkflowStatus(chapter)} />
            <WorkflowTag label="传" status={publishWorkflowStatus(chapter)} />
          </div>
        </div>
      </div>
    </div>
  );
}

/**
 * 修复后的 WorkflowTag：文字垂直居中，颜色极淡化
 */
function WorkflowTag({ label, status }) {
  const statusConfig = {
    pending: {
      text: 'text-slate-300',
      line: 'bg-slate-100',
      bg: 'bg-slate-50/40'
    },
    in_progress: {
      text: 'text-orange-300',
      line: 'bg-orange-200',
      bg: 'bg-orange-50/40'
    },
    completed: {
      text: 'text-emerald-400',
      line: 'bg-emerald-50',
      bg: 'bg-emerald-50/50',
      lineStrong: 'bg-emerald-300' // 完成状态线稍微明显一点点
    }
  };

  const style = statusConfig[status];

  return (
    <div className={`relative flex-1 h-6 flex items-center justify-center rounded-sm transition-all duration-300 ${style.bg}`}>
      {/* 文字：绝对垂直居中 */}
      <span className={`text-[10px] font-bold tracking-widest ${style.text} z-10`}>
        {label}
      </span>
      {/* 底部状态线：绝对定位贴底 */}
      <div className={`absolute bottom-0 left-0 right-0 h-[2.5px] rounded-b-sm ${style.lineStrong || style.line}`} />
    </div>
  );
}
