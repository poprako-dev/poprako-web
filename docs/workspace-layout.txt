// --- 子组件: 工作流标签 ---
function WorkflowTag({ label, status }) {
  const cfg = STATUS_CONFIG[status] ?? STATUS_CONFIG.pending;
  return (
    <div className={`relative flex-1 h-6 flex items-center justify-center rounded-sm transition-all duration-300 ${cfg.bg}`}>

      <div className={`absolute bottom-0 left-0 right-0 h-[2.5px] rounded-b-sm ${cfg.line}`} />
    </div>
  );
}
