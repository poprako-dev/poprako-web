import React, { useState, useEffect, useRef } from 'react';
import { X, Info } from 'lucide-react';

// --- 固定快捷键数据 ---
const FIXED_SHORTCUTS = [
  { label: '新建框内标签', value: '左键点击图片' },
  { label: '新建框外标签', value: '右键点击图片' },
  { label: '切换框内框外', value: '中键点击标签' },
  { label: '删除标签', value: '右键点击标签' },
];

// --- 可配置快捷键数据 ---
const INITIAL_CONFIGURABLE = [
  { id: 'prev-tab', label: '上一个标签', keys: ['Shift', 'Tab'] },
  { id: 'next-tab', label: '下一个标签', keys: ['Tab'] },
  { id: 'prev-page', label: '上一页', keys: ['Control', 'ArrowLeft'] },
  { id: 'next-page', label: '下一页', keys: ['Control', 'ArrowRight'] },
  { id: 'save', label: '保存修改', keys: ['Control', 's'] },
  { id: 'undo', label: '撤销操作', keys: ['Control', 'z'] },
];

const STORAGE_KEY = 'translator_settings_v4';

const App = () => {
  const [shortcuts, setShortcuts] = useState([]);
  const [isRecording, setIsRecording] = useState(null); 
  const recordedKeys = useRef(new Set());

  useEffect(() => {
    const saved = localStorage.getItem(STORAGE_KEY);
    setShortcuts(saved ? JSON.parse(saved) : INITIAL_CONFIGURABLE);
  }, []);

  const saveToStorage = (updated) => {
    setShortcuts(updated);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
  };

  useEffect(() => {
    const handleKeyDown = (e) => {
      if (isRecording) {
        e.preventDefault();
        recordedKeys.current.add(e.key);
        return;
      }
    };

    const handleKeyUp = () => {
      if (isRecording) {
        const keysArray = Array.from(recordedKeys.current);
        if (keysArray.length > 0) {
          const updated = shortcuts.map(s => 
            s.id === isRecording ? { ...s, keys: keysArray } : s
          );
          saveToStorage(updated);
          setIsRecording(null);
          recordedKeys.current.clear();
        }
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    window.addEventListener('keyup', handleKeyUp);
    return () => {
      window.removeEventListener('keydown', handleKeyDown);
      window.removeEventListener('keyup', handleKeyUp);
    };
  }, [isRecording, shortcuts]);

  const removeKey = (id) => {
    const updated = shortcuts.map(s => s.id === id ? { ...s, keys: null } : s);
    saveToStorage(updated);
  };

  const formatKeys = (keys) => {
    if (!keys || keys.length === 0) return '';
    return keys.map(k => (k === ' ' ? 'Space' : k)).join('+');
  };

  return (
    <div className="min-h-screen bg-gray-100 flex items-center justify-center p-4 font-sans text-gray-800">
      <div className="bg-white w-full max-w-4xl rounded shadow-lg overflow-hidden">
        
        {/* Header */}
        <div className="flex justify-between items-center px-6 py-3 border-b border-gray-100">
          <span className="font-bold text-gray-700">翻译器快捷键设置</span>
          <button className="text-gray-400 hover:text-gray-600 transition-colors">
            <X size={20} />
          </button>
        </div>

        <div className="p-8">
          {/* Section 1: 固定功能展示 (1:1 比例) */}
          <div className="grid grid-cols-2 gap-x-16 gap-y-5 mb-10 pb-4 border-b border-dashed border-gray-100">
            {FIXED_SHORTCUTS.map((item, index) => (
              <div key={index} className="grid grid-cols-2 items-center text-sm">
                <span className="text-gray-400">{item.label}</span>
                <span className="text-gray-600 font-medium">{item.value}</span>
              </div>
            ))}
          </div>

          {/* Section 2: 快捷键配置 (1:1 比例) */}
          <div className="grid grid-cols-2 gap-x-16 gap-y-6">
            {shortcuts.map((s) => (
              <div key={s.id} className="grid grid-cols-2 items-center">
                <span className="text-gray-500 text-sm">{s.label}</span>
                <div className="flex gap-2">
                  <div 
                    onClick={() => setIsRecording(s.id)}
                    className={`
                      flex-1 h-9 px-3 flex items-center rounded border text-sm transition-all select-none cursor-pointer
                      ${isRecording === s.id 
                        ? 'border-blue-400 bg-blue-50 text-blue-600 ring-2 ring-blue-50' 
                        : 'border-gray-100 bg-gray-50/50 text-gray-400 hover:border-gray-200'}
                    `}
                  >
                    {isRecording === s.id ? '请按下按键...' : formatKeys(s.keys)}
                  </div>
                </div>
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
};

export default App;
