// ==========================================

const TRANSLATOR_NAV = { 
  id: 'translator', 
  label: '星空汉化组', 
  icon: Globe2, 
  path: '/translator' 
};

const MAIN_NAV = [
  { id: 'dashboard', label: '仪表盘', icon: LayoutDashboard, path: '/dashboard' },
  { id: 'community', label: '社区大厅', icon: Users, path: '/community' },
  { id: 'library', label: '资源文库', icon: BookOpen, path: '/library' },
  { id: 'notifications', label: '消息中心', icon: Bell, path: '/notifications' },

  id: 'settings',

};

// ==========================================
// 2. Component Implementations
// ==========================================

/**
 * AppLogoRow - Fixed: Absolute centering using w-14 container
 */
const AppLogoRow = ({ title }) => {
  return (
    <div className="flex h-16 w-full items-center text-[#2e3c33] relative">
      {/* Icon Container: Fixed w-14 ensures it sits exactly in the 56px collapsed sidebar */}
      <div className="flex w-14 shrink-0 items-center justify-center">
        <Trees className="h-7 w-7" />
      </div>
      
      {/* Text Label: Placed outside the 14-unit icon space */}
      <span className="whitespace-nowrap text-lg font-bold tracking-tight opacity-0 transition-opacity duration-300 group-hover:opacity-100 absolute left-14">
        {title}
      </span>
    </div>
  );
};

/**
 * SidebarItem - Optimized for pixel-perfect centering in collapsed state
 */
const SidebarItem = ({ icon: Icon, label, isActive, onClick }) => {
  return (
    <button
      onClick={onClick}
      className={`
        group/item relative flex h-11 w-full items-center transition-all duration-200 ease-in-out
        outline-none overflow-hidden whitespace-nowrap
        ${isActive 
          ? "text-[#2e5c33]" 
          : "text-[#7a8c80] hover:text-[#2e3c33]"
        }
      `}
    >
      {/* Background/Border Overlay: Managed separately to avoid pushing the icon container */}
      <div className={`absolute inset-y-0 left-1 right-1 rounded-lg transition-colors duration-200 
        ${isActive ? "bg-[#e3f2e5] border border-[#c8e6c9]" : "group-hover/item:bg-[#f0f4f1]"}
      `} />

      {/* Icon Container: Forced w-14 with z-index to stay above background */}
      <span className="relative z-10 flex w-14 shrink-0 items-center justify-center">
        <Icon
          size={20}
          className={`
            transition-transform duration-300 ease-[cubic-bezier(0.175,0.885,0.32,1.275)]
            group-hover/item:scale-110 group-hover/item:-rotate-6
          `}
        />
      </span>

      {/* Text Label: Positioned at exactly the 14th unit (56px) marker */}
      <span
        className={`
          relative z-10 text-sm font-medium tracking-wide
          -translate-x-2 opacity-0
          transition-all delay-75 duration-300
          group-hover:translate-x-2 group-hover:opacity-100
        `}
      >
        {label}
      </span>
    </button>
  );
};

/**
 * SidebarContainer - Physical Layout
 */
const SidebarContainer = ({ header, footer, children }) => {
  return (
    <nav className="group peer fixed left-0 top-0 z-50 flex h-screen w-14 flex-col border-r border-gray-200 bg-white transition-[width] duration-400 ease-in-out hover:w-56 hover:shadow-md">
      {/* Header section */}
      <div className="flex-shrink-0">
        {header}
      </div>

      {/* Main content area: Removing all horizontal padding to ensure w-14 items are centered */}
      <div className="flex-1 overflow-y-auto overflow-x-hidden no-scrollbar">
        {children}
      </div>

      {/* Footer section */}
      <div className="mt-auto flex-shrink-0 border-t border-gray-50 bg-gray-50/30 py-2">
        {footer}
      </div>
    </nav>
  );
};

/**
 * AppNavSidebar - Business Logic Layer
 */
const AppNavSidebar = ({ currentPath, onNavigate }) => {
  return (
    <SidebarContainer
      header={<AppLogoRow title="PopRaKo" />}
      footer={
        <SidebarItem
          icon={FOOTER_NAV.icon}
          label={FOOTER_NAV.label}
          isActive={currentPath === FOOTER_NAV.path}
          onClick={() => onNavigate(FOOTER_NAV.path)}
        />
      }
    >
      <div className="flex flex-col gap-1">
        {/* Translator Section (Single Slot) */}
        <div className="mb-2 pb-2 border-b border-gray-50">
          <SidebarItem
            icon={TRANSLATOR_NAV.icon}
            label={TRANSLATOR_NAV.label}
            isActive={currentPath === TRANSLATOR_NAV.path}
            onClick={() => onNavigate(TRANSLATOR_NAV.path)}
          />
        </div>

        {/* Navigation Tabs (Multiple Slots) */}
        {MAIN_NAV.map((item) => (
          <SidebarItem
            key={item.id}
            icon={item.icon}
            label={item.label}
            isActive={currentPath === item.path}
            onClick={() => onNavigate(item.path)}
          />
        ))}
      </div>
    </SidebarContainer>
  );
};

          </header>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
