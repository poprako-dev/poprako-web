import React, { useState, useRef, useEffect } from 'react';
import { Plus } from 'lucide-react';

// --- Types ---
interface CharItem {
  id: string;
  text: string;
  isFavorite: boolean;
}

type Mode = 'select' | 'delete';

// --- Main Component ---
export default function App() {
  // --- State ---
  const [mode, setMode] = useState<Mode>('select');
  const [chars, setChars] = useState<CharItem[]>([
    { id: '1', text: '✧', isFavorite: true },
    { id: '2', text: '(｡♥‿♥｡)', isFavorite: false },
    { id: '3', text: '★', isFavorite: false },
    { id: '4', text: '→', isFavorite: true },
    { id: '5', text: '【】', isFavorite: false },
    { id: '6', text: 'ツ', isFavorite: false },
    { id: '7', text: '§', isFavorite: false },
  ]);
  
  const [isAdding, setIsAdding] = useState(false);
  const [newCharText, setNewCharText] = useState('');
  const textareaRef = useRef<HTMLTextAreaElement>(null);

  // --- Effects ---
  useEffect(() => {
    if (isAdding && textareaRef.current) {
      textareaRef.current.focus();
    }
  }, [isAdding]);

  // --- Handlers ---
  const handleToggleFavorite = (id: string) => {
    if (mode !== 'select') return;
    setChars(chars.map(c => 
      c.id === id ? { ...c, isFavorite: !c.isFavorite } : c
    ));
  };

  const handleDelete = (id: string) => {
    if (mode !== 'delete') return;
    setChars(chars.filter(c => c.id !== id));
  };

  const submitNewChar = () => {
    const trimmed = newCharText.trim();
    if (trimmed) {
      setChars([...chars, { id: Date.now().toString(), text: trimmed, isFavorite: false }]);
    }
    setIsAdding(false);
    setNewCharText('');
  };

  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      submitNewChar();
    }
    if (e.key === 'Escape') {
      setIsAdding(false);
      setNewCharText('');
    }
  };

  // --- Render Helpers ---
  return (
    <div className="min-h-screen bg-white flex items-start justify-center pt-24 pb-12 font-sans selection:bg-emerald-100 selection:text-emerald-900">
      <div className="w-full max-w-2xl px-6">
        
        {/* Header Tabs */}
        <div className="flex justify-center mb-10">
          <div className="inline-flex items-center p-1 bg-[#F4F5F7] rounded-[14px]">
            <button
              onClick={() => { setMode('select'); setIsAdding(false); }}
              className={`
                relative px-6 py-2 text-[15px] rounded-[10px] transition-all duration-300 ease-out outline-none
                ${mode === 'select' 
                  ? 'bg-white text-[#2C313D] shadow-[0_2px_10px_rgba(0,0,0,0.03)] font-medium tracking-wide' 
                  : 'text-[#94A3B8] hover:text-[#64748B] hover:bg-slate-200/50'}
              `}
            >
              优选
            </button>
            <button
              onClick={() => { setMode('delete'); setIsAdding(false); }}
              className={`
                relative px-6 py-2 text-[15px] rounded-[10px] transition-all duration-300 ease-out outline-none
                ${mode === 'delete' 
                  ? 'bg-white text-[#2C313D] shadow-[0_2px_10px_rgba(0,0,0,0.03)] font-medium tracking-wide' 
                  : 'text-[#94A3B8] hover:text-[#64748B] hover:bg-slate-200/50'}
              `}
            >
              删除
            </button>
          </div>
        </div>

        {/* Char Grid */}
        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-5 gap-3 relative">
          
          {chars.map((char) => (
            <button
              key={char.id}
              onClick={() => mode === 'select' ? handleToggleFavorite(char.id) : handleDelete(char.id)}
              className={`
                group relative flex items-center justify-center h-14 rounded-xl text-base transition-all duration-300 ease-out outline-none
                active:scale-[0.96] overflow-hidden
                ${mode === 'select' 
                  ? char.isFavorite 
                    ? 'bg-[#F0FDF4] text-[#059669]' // 极浅绿
                    : 'bg-[#F8F9FA] text-[#475569] hover:bg-[#F1F3F5]' // 淡雅灰
                  : 'bg-[#F8F9FA] text-[#475569] hover:bg-[#FEF2F2] hover:text-[#DC2626]'} // 删除模式悬浮变浅红
              `}
            >
              <span className="relative z-10 transition-transform duration-300 group-hover:scale-105">
                {char.text}
              </span>
            </button>
          ))}

          {/* Add Button / Textarea (Only in Select Mode) */}
          {mode === 'select' && (
            <div className="relative h-14">
              {isAdding ? (
                <textarea
                  ref={textareaRef}
                  value={newCharText}
                  onChange={(e) => setNewCharText(e.target.value)}
                  onKeyDown={handleKeyDown}
                  onBlur={submitNewChar}
                  placeholder="..."
                  className="
                    absolute inset-0 w-full h-full p-0 m-0 text-center flex items-center justify-center
                    bg-white text-[#2C313D] text-base rounded-xl outline-none resize-none
                    border border-[#E2E8F0] shadow-[0_4px_12px_rgba(0,0,0,0.04)]
                    transition-all duration-300 pt-[14px]
                    placeholder:text-[#CBD5E1]
                  "
                />
              ) : (
                <button
                  onClick={() => setIsAdding(true)}
                  className="
                    w-full h-full flex items-center justify-center rounded-xl bg-[#F8F9FA]/60 text-[#CBD5E1]
                    transition-all duration-300 hover:bg-[#F1F3F5] hover:text-[#94A3B8] outline-none
                    active:scale-[0.96]
                  "
                >
                  <Plus strokeWidth={2.5} className="w-5 h-5 transition-transform duration-300" />
                </button>
              )}
            </div>
          )}
          
        </div>
        
      </div>
    </div>
  );
}
