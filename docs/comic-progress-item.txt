function ComicProgressItem({ chapter, isSelected, onClick }) {
  // 极窄状态配色
  const getStatusColor = (status) => {
    if (status === 'finished') return 'bg-emerald-500';
    if (status === 'raw') return 'bg-stone-300';
    return 'bg-amber-500';
  };

  return (
    <div 
      onClick={onClick}
      className={`
        group relative w-full flex flex-col justify-between 
        bg-white hover:bg-stone-50/40
        border border-stone-200 hover:border-stone-300
        rounded-sm overflow-hidden transition-all duration-150 cursor-pointer
        ${isSelected ? 'border-stone-400 shadow-[0_2px_6px_rgba(0,0,0,0.015)]' : 'shadow-[0_1px_2px_rgba(0,0,0,0.01)]'}
      `}
    >
      {/* 第一行：左对齐序号与标题，右对齐时钟与时间 */}
      <div className="flex items-center justify-between pt-2 pb-1.5 px-3 min-w-0 w-full">
        {/* 左对齐：#序号 + 标题 */}
        <div className="flex items-center gap-1.5 min-w-0 flex-1">
          <span className="text-[10px] font-bold text-stone-400 font-mono shrink-0">#{chapter.index}</span>
          <span className="text-[11px] font-bold text-stone-800 truncate">
            {chapter.mangaTitle} - {chapter.title}
          </span>
        </div>

        {/* 右对齐：时钟 + 时间 + 更精微的指示圆点 */}
        <div className="flex items-center gap-1 shrink-0 ml-3">
          <Clock className="w-3 h-3 text-stone-400" />
          <span className="text-[9px] text-stone-400 font-mono">{chapter.updatedAt}</span>
          <div className={`w-1.5 h-1.5 rounded-full ml-1 ${getStatusColor(chapter.status)}`} />
        </div>
      </div>

      {/* 第二行：100% 宽度通栏内阴影暗槽（无任何颜色状态区分，极致纯净平等） */}
      <div className="w-full bg-stone-100/40 border-t border-stone-200/30 shadow-[inset_0_1.5px_3px_rgba(0,0,0,0.035)]">
        <div className="grid grid-cols-6 text-center select-none text-[10px] font-semibold font-mono text-stone-600/90">
          {Object.entries(WORKFLOW_STEPS).map(([key, config]) => {
            const stepData = chapter.workflow[key] || { count: 0, status: 'pending' };
            
            return (
              <div 
                key={key}
                className="py-1 flex items-center justify-center"
              >
                <div className="flex items-center gap-0.5">
                  <span className="tracking-wide">{config.label}</span>
                  {stepData.count > 0 && (
                    <span className="
                      text-[8px] px-1 py-0.2 ml-0.5 rounded-full font-bold leading-none
                      bg-stone-200 text-stone-550 scale-90 origin-center
                    ">
                      +{stepData.count}
                    </span>
                  )}
                </div>
              </div>
            );
          })}
        </div>
      </div>
    </div>
  );
}

function WorkflowEditor({ chapter, onClose, onSave }) {
  const [editedChapter, setEditedChapter] = useState(JSON.parse(JSON.stringify(chapter)));

  const handleCountChange = (stepKey, value) => {
    const val = parseInt(value) || 0;
    setEditedChapter(prev => {
      const updated = { ...prev };
      updated.workflow[stepKey].count = val;
      return updated;
    });
  };

  const handleStatusChange = (stepKey, status) => {
    setEditedChapter(prev => {
      const updated = { ...prev };
      updated.workflow[stepKey].status = status;
      return updated;
    });
  };

  const handleSave = () => {
    const steps = Object.keys(editedChapter.workflow);
    const doneCount = steps.filter(s => editedChapter.workflow[s].status === 'done').length;
    const progress = Math.round((doneCount / steps.length) * 100);
    
    let overallStatus = 'raw';
    if (progress === 100) overallStatus = 'finished';
    else if (progress > 80) overallStatus = 'qc';
    else if (progress > 50) overallStatus = 'typesetting';
    else if (progress > 20) overallStatus = 'translate';

    onSave({
      ...editedChapter,
      progress,
      status: overallStatus,
    });
  };

  return (
    <div className="w-full bg-white border border-stone-200/60 rounded-lg shadow-[0_4px_20px_rgba(0,0,0,0.015)] overflow-hidden transition-all duration-300">
      <div className="flex items-center justify-between px-3 py-2 border-b border-stone-100 bg-stone-50/40">
        <div>
          <span className="text-[9px] text-stone-400 font-mono font-medium">#{chapter.index} EDITING</span>
          <h3 className="text-xs font-semibold text-stone-700 truncate max-w-[160px] mt-0.5">
            {chapter.title}
          </h3>
        </div>
        <button 
          onClick={onClose}
          className="p-1 rounded-full text-stone-400 hover:bg-stone-50 hover:text-stone-600 transition-colors"
        >
          <X className="w-3 h-3" />
        </button>
      </div>

      <div className="p-3 space-y-2">
        {Object.entries(WORKFLOW_STEPS).map(([key, config]) => {
          const stepData = editedChapter.workflow[key];
          return (
            <div key={key} className="flex items-center justify-between py-0.5 border-b border-stone-50 last:border-0">
              <div>
                <span className="text-xs font-medium text-stone-700">{config.name}</span>
                <span className="text-[9px] text-stone-400 font-mono ml-1">({config.label})</span>
              </div>

              <div className="flex items-center gap-1.5">
                <div className="flex items-center gap-0.5">
                  <span className="text-[9px] text-stone-400 font-mono">人:</span>
                  <input 
                    type="number" 
                    min="0"
                    max="9"
                    value={stepData.count}
                    onChange={(e) => handleCountChange(key, e.target.value)}
                    className="w-7 px-1 py-0.5 border border-stone-200 rounded-xs font-mono text-[10px] text-center focus:outline-hidden"
                  />
                </div>

                <div className="flex p-0.5 bg-stone-100 rounded-sm gap-0.5">
                  {['pending', 'active', 'done'].map((status) => (
                    <button
                      key={status}
                      type="button"
                      onClick={() => handleStatusChange(key, status)}
                      className={`
                        px-1 text-[8.5px] rounded-[1px] transition-all
                        ${stepData.status === status 
                          ? 'bg-white text-stone-800 shadow-xs font-medium' 
                          : 'text-stone-400'
                        }
                      `}
                    >
                      {status === 'pending' && '待'}
                      {status === 'active' && '行'}
                      {status === 'done' && '完'}
                    </button>
                  ))}
                </div>
              </div>
            </div>
          );
        })}
      </div>

      <div className="p-2 border-t border-stone-100 bg-stone-50/30 flex gap-1.5">
        <button 
          onClick={onClose}
          className="flex-1 py-1 border border-stone-200 text-[10px] text-stone-500 hover:text-stone-700 rounded-md transition-colors"
        >
          取消
        </button>
        <button 
          onClick={handleSave}
          className="flex-1 py-1 bg-stone-850 text-white text-[10px] hover:bg-stone-900 rounded-md transition-colors"
        >
          保存
        </button>
      </div>
    </div>
  );
}

export default function App() {
  const [chapters, setChapters] = useState(INITIAL_CHAPTERS);
  const [selectedManga, setSelectedManga] = useState('All');
  const [searchQuery, setSearchQuery] = useState('');
  const [activeTab, setActiveTab] = useState('all');
  const [editingChapter, setEditingChapter] = useState(null);
  const [showAddModal, setShowAddModal] = useState(false);

  const mangaList = useMemo(() => {
    return ['All', ...new Set(chapters.map(c => c.mangaTitle))];
  }, [chapters]);

  const [newChapter, setNewChapter] = useState({
    mangaTitle: '葬送的芙莉莲',
    index: '',
    title: '',
    workflow: {
      raw: { count: 0, status: 'pending' },
      translate: { count: 0, status: 'pending' },
      proof: { count: 0, status: 'pending' },
      typeset: { count: 0, status: 'pending' },
      qc: { count: 0, status: 'pending' },
      upload: { count: 0, status: 'pending' }
    }
  });

  const filteredChapters = useMemo(() => {
    return chapters.filter(chap => {
      const matchManga = selectedManga === 'All' || chap.mangaTitle === selectedManga;
      const matchSearch = chap.title.toLowerCase().includes(searchQuery.toLowerCase()) || 
                          chap.index.includes(searchQuery);
      const matchStatus = 
        activeTab === 'all' || 
        (activeTab === 'active' && chap.status !== 'finished') ||
        (activeTab === 'finished' && chap.status === 'finished');
      return matchManga && matchSearch && matchStatus;
    });
  }, [chapters, selectedManga, searchQuery, activeTab]);

  const handleUpdateChapter = (updated) => {
    setChapters(prev => prev.map(c => c.id === updated.id ? updated : c));
    setEditingChapter(null);
  };

  const handleAddChapter = (e) => {
    e.preventDefault();
    if (!newChapter.index || !newChapter.title) return;

    const added = {
      ...newChapter,
      id: Date.now().toString(),
      status: 'raw',
      progress: 0,
      updatedAt: '2026/5/17',
      workflow: {
        raw: { count: 1, status: 'active' },
        translate: { count: 0, status: 'pending' },
        proof: { count: 0, status: 'pending' },
        typeset: { count: 0, status: 'pending' },
        qc: { count: 0, status: 'pending' },
        upload: { count: 0, status: 'pending' }
      }
    };

    setChapters(prev => [added, ...prev]);
    setShowAddModal(false);
    setNewChapter({
      mangaTitle: '葬送的芙莉莲',
      index: '',
      title: '',
      workflow: {
        raw: { count: 0, status: 'pending' },
        translate: { count: 0, status: 'pending' },
        proof: { count: 0, status: 'pending' },
        typeset: { count: 0, status: 'pending' },
        qc: { count: 0, status: 'pending' },
        upload: { count: 0, status: 'pending' }
      }
    });
  };

  return (
    <div className="min-h-screen bg-[#fafaf9] text-stone-700 flex flex-col antialiased">
      {/* 极简顶栏 */}
      <header className="border-b border-stone-200/30 bg-white/60 backdrop-blur-md sticky top-0 z-30">
        <div className="max-w-3xl mx-auto px-4 h-12 flex items-center justify-between">
          <div className="flex items-center gap-2">
            <div className="w-5 h-5 bg-stone-800 rounded-sm flex items-center justify-center">
              <FolderKanban className="w-3.5 h-3.5 text-stone-50" />
            </div>
            <h1 className="text-xs font-semibold tracking-tight text-stone-800">Manga Pipeline</h1>
          </div>

          <button 
            onClick={() => setShowAddModal(true)}
            className="flex items-center gap-1 px-2.5 py-1 bg-stone-850 hover:bg-stone-900 text-white text-[11px] rounded-md transition-colors"
          >
            <Plus className="w-3.5 h-3.5" />
            <span>新增条目</span>
          </button>
        </div>
      </header>

      {/* 主面板 */}
      <main className="flex-1 max-w-3xl w-full mx-auto px-4 py-4 flex flex-col md:flex-row gap-5">
        
        {/* 左侧列表 */}
        <div className="flex-1 space-y-3">
          
          {/* 筛选控制器 */}
          <div className="flex flex-col sm:flex-row gap-2 items-start sm:items-center justify-between">
            <div className="flex p-0.5 bg-stone-200/40 rounded-lg">
              {['all', 'active', 'finished'].map(tab => (
                <button
                  key={tab}
                  onClick={() => setActiveTab(tab)}
                  className={`px-2.5 py-0.5 text-[10px] rounded-md transition-all ${activeTab === tab ? 'bg-white text-stone-800 shadow-xs font-medium' : 'text-stone-400 hover:text-stone-600'}`}
                >
                  {tab === 'all' && '全部'}
                  {tab === 'active' && '进行中'}
                  {tab === 'finished' && '已发布'}
                </button>
              ))}
            </div>

            <div className="relative w-full sm:w-36">
              <Search className="absolute left-2 top-1/2 -translate-y-1/2 w-3 h-3 text-stone-400" />
              <input
                type="text"
                placeholder="搜索话数/名字..."
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                className="w-full pl-6 pr-2 py-0.5 bg-stone-100/50 border border-stone-200/50 rounded-md text-[10px] focus:outline-hidden focus:border-stone-400 focus:bg-white transition-all"
              />
            </div>
          </div>

          {/* 作品过滤标签 */}
          <div className="flex flex-wrap items-center gap-1">
            {mangaList.map(manga => (
              <button
                key={manga}
                onClick={() => setSelectedManga(manga)}
                className={`
                  px-2 py-0.5 text-[9px] rounded-md border transition-all duration-150
                  ${selectedManga === manga 
                    ? 'bg-stone-800 text-stone-50 border-stone-800' 
                    : 'bg-white text-stone-400 border-stone-200/60 hover:border-stone-350'
                  }
                `}
              >
                {manga === 'All' ? '全部项目' : manga}
              </button>
            ))}
          </div>

          {/* 列表渲染 */}
          <div className="space-y-2">
            {filteredChapters.length > 0 ? (
              filteredChapters.map(chap => (
                <ComicProgressItem
                  key={chap.id}
                  chapter={chap}
                  isSelected={editingChapter?.id === chap.id}
                  onClick={() => setEditingChapter(chap)}
                />
              ))
            ) : (
              <div className="text-center py-8 bg-stone-50/20 border border-stone-200/20 rounded-xl">
                <Info className="w-3.5 h-3.5 text-stone-300 mx-auto mb-1.5" />
                <span className="text-[10px] text-stone-400">暂无数据记录</span>
              </div>
            )}
          </div>
        </div>

        {/* 右侧编辑 */}
        <div className="w-full md:w-56 shrink-0">
          {editingChapter ? (
            <WorkflowEditor
              chapter={editingChapter}
              onClose={() => setEditingChapter(null)}
              onSave={handleUpdateChapter}
            />
          ) : (
            <div className="p-4 bg-white/40 border border-stone-200/40 rounded-lg text-center space-y-2">
              <div className="w-7 h-7 bg-stone-100/60 rounded-full flex items-center justify-center mx-auto text-stone-400">
                <SlidersHorizontal className="w-3.5 h-3.5" />
              </div>
              <div>
                <h4 className="text-[10px] font-medium text-stone-600">流程微调</h4>
                <p className="text-[9px] text-stone-400 leading-normal max-w-[140px] mx-auto mt-0.5">
                  点击左侧漫画卡片，在这里直接精准控制人员分工与完成状态。
                </p>
              </div>
            </div>
          )}
        </div>
      </main>

      {/* 新建模态 */}
      {showAddModal && (
        <div className="fixed inset-0 bg-stone-900/10 backdrop-blur-xs z-50 flex items-center justify-center p-4">
          <div className="bg-white w-full max-w-xs rounded-lg border border-stone-200/80 shadow-[0_12px_30px_rgba(28,25,23,0.04)] overflow-hidden">
            <div className="px-3.5 py-2 border-b border-stone-100 flex items-center justify-between">
              <h3 className="text-xs font-semibold text-stone-700 flex items-center gap-1">
                <Sparkles className="w-3 h-3 text-amber-500" />
                新建进度条目
              </h3>
              <button onClick={() => setShowAddModal(false)} className="p-1 rounded-full text-stone-400 hover:bg-stone-50">
                <X className="w-3 h-3" />
              </button>
            </div>

            <form onSubmit={handleAddChapter} className="p-3.5 space-y-2.5">
              <div>
                <label className="block text-[8px] font-medium text-stone-400 uppercase mb-0.5">所属项目</label>
                <select 
                  value={newChapter.mangaTitle}
                  onChange={(e) => setNewChapter(prev => ({ ...prev, mangaTitle: e.target.value }))}
                  className="w-full px-2 py-0.5 bg-stone-50 border border-stone-200/60 rounded-md text-[10px] focus:outline-hidden"
                >
                  <option value="葬送的芙莉莲">葬送的芙莉莲</option>
                  <option value="迷宫饭">迷宫饭</option>
                </select>
              </div>

              <div className="grid grid-cols-3 gap-2">
                <div>
                  <label className="block text-[8px] font-medium text-stone-400 uppercase mb-0.5">章节</label>
                  <input 
                    type="text" 
                    placeholder="120"
                    required
                    value={newChapter.index}
                    onChange={(e) => setNewChapter(prev => ({ ...prev, index: e.target.value }))}
                    className="w-full px-2 py-0.5 bg-stone-50 border border-stone-200/60 rounded-md font-mono text-[10px] focus:outline-hidden"
                  />
                </div>
                <div className="col-span-2">
                  <label className="block text-[8px] font-medium text-stone-400 uppercase mb-0.5">标题</label>
                  <input 
                    type="text" 
                    placeholder="输入章节名"
                    required
                    value={newChapter.title}
                    onChange={(e) => setNewChapter(prev => ({ ...prev, title: e.target.value }))}
                    className="w-full px-2 py-0.5 bg-stone-50 border border-stone-200/60 rounded-md text-[10px] focus:outline-hidden"
                  />
                </div>
              </div>

              <div className="pt-1.5 flex gap-1.5">
                <button 
                  type="button"
                  onClick={() => setShowAddModal(false)}
                  className="flex-1 py-1 border border-stone-200 text-[10px] text-stone-500 rounded-md hover:text-stone-800"
                >
                  取消
                </button>
                <button 
                  type="submit"
                  className="flex-1 py-1 bg-stone-900 text-white text-[10px] rounded-md hover:bg-stone-850"
                >
                  创建
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
