import React, { useState } from 'react';
import { 
  Trees, 
  Settings, 
  LayoutDashboard, 
  Users, 
  BookOpen, 
  Globe2, 
  Bell, 
  Check,
  ChevronDown
} from 'lucide-react';

// ==========================================
// 1. 数据配置
// ==========================================
const TRANSLATORS = [
  { id: 'xk', name: '星空汉化组', short: 'XK', desc: '专注二次元翻译' },
  { id: 'mh', name: '迷幻汉化组', short: 'MH', desc: '硬核动作游戏汉化' },
  { id: 'sy', name: '深渊汉化组', short: 'SY', desc: '独立游戏爱好者' },
];

const MAIN_NAV = [
  { id: 'dashboard', label: '仪表盘', icon: LayoutDashboard, path: '/dashboard' },
  { id: 'community', label: '社区大厅', icon: Users, path: '/community' },
  { id: 'library', label: '资源文库', icon: BookOpen, path: '/library' },
  { id: 'notifications', label: '消息中心', icon: Bell, path: '/notifications' },
];

const FOOTER_NAV = { id: 'settings', label: '全局设置', icon: Settings, path: '/settings' };

// ==========================================
// 2. 异构组件实现 (精准对齐版)
// ==========================================

/**
 * TranslatorList - 在右侧悬浮显示的列表 (修复了 Hover 丢失问题)
 */
const TranslatorList = ({ isOpen, activeId, onSelect }) => {
  if (!isOpen) return null;

  return (
    // 核心修复 1：利用 left-full 和 pl-3 (Padding Left) 创造一个隐形的“悬停桥梁”。
    // 鼠标在经过按钮到列表之间的空隙时，实际上是在这个透明的 Padding 上滑动，因此绝对不会触发 onMouseLeave！
    <div className="absolute left-full top-0 pl-3 z-[100] animate-in fade-in slide-in-from-left-2 duration-200">
      <div className="w-64 bg-white border border-gray-100 rounded-2xl shadow-[0_20px_50px_rgba(0,0,0,0.1)] py-3">
        <div className="px-5 py-2 mb-2">
          <h4 className="text-[11px] font-black text-gray-400 uppercase tracking-widest">切换汉化组</h4>
        </div>
        <div className="space-y-1 px-2">
          {TRANSLATORS.map((t) => (
            <button
              key={t.id}
              onClick={() => onSelect(t)}
              className={`
                w-full flex items-center gap-4 px-4 py-3 rounded-xl transition-all
                ${t.id === activeId 
                  ? 'bg-green-50 text-green-800' 
                  : 'text-gray-500 hover:bg-gray-50 hover:text-gray-900'}
              `}
            >
              <div className={`w-10 h-10 rounded-lg flex items-center justify-center font-black text-sm
                ${t.id === activeId ? 'bg-green-600 text-white' : 'bg-gray-100 text-gray-400'}`}>
                {t.short}
              </div>
              <div className="flex flex-col items-start text-left">
                <span className="text-sm font-bold">{t.name}</span>
                <span className="text-[10px] opacity-60">{t.desc}</span>
              </div>
              {t.id === activeId && <Check size={16} className="ml-auto text-green-600" />}
            </button>
          ))}
        </div>
      </div>
    </div>
  );
};

/**
 * TranslatorButton - 异构按钮 (严格限制物理尺寸以保居中)
 */
const TranslatorButton = ({ activeTranslator, isOpen, onClick }) => {
  return (
    <div className="relative w-full h-16 group/trans">
      <button
        onClick={onClick}
        className="w-full h-full flex items-center outline-none"
      >
        {/* 核心修复 2：外层没有任何 padding，强制图标容器为严格的 w-14，内部卡片 w-10 居中 */}
        <div className="relative z-10 w-14 shrink-0 h-full flex items-center justify-center">
          <div className={`
            flex items-center justify-center w-11 h-11 rounded-xl transition-all duration-300
            ${isOpen ? 'bg-green-600 shadow-md scale-105' : 'bg-[#2e5c33] group-hover/trans:bg-[#3a7340]'}
          `}>
            <Globe2 size={22} className={`text-white transition-transform duration-500 ${isOpen ? 'rotate-12' : ''}`} />
          </div>
        </div>

        {/* 文本内容：绝对定位脱离文档流，永远不挤压图标空间 */}
        <div className="absolute left-14 right-2 h-full flex flex-col justify-center opacity-0 group-hover:opacity-100 transition-opacity duration-300">
          <div className={`
            px-3 py-2 rounded-xl transition-colors duration-300 h-11 flex flex-col justify-center
            ${isOpen ? 'bg-green-50' : 'hover:bg-gray-50'}
          `}>
             <span className="text-[10px] font-bold text-gray-400 leading-none mb-1">当前汉化组</span>
             <div className="flex items-center gap-1.5">
               <span className="text-sm font-bold text-[#2e3c33] tracking-wide truncate">{activeTranslator.name}</span>
               <ChevronDown size={14} className={`text-gray-400 transition-transform duration-300 shrink-0 ${isOpen ? 'rotate-180 text-green-600' : ''}`} />
             </div>
          </div>
        </div>
      </button>

      {/* 弹出的子列表 */}
      <TranslatorList 
        isOpen={isOpen}
        activeId={activeTranslator.id}
        onSelect={onClick} // 这里将 onClick 作为选择后的回调
      />
    </div>
  );
};

/**
 * 标准 SidebarItem
 */
const SidebarItem = ({ icon: Icon, label, isActive, onClick }) => {
  return (
    <button
      onClick={onClick}
      className={`
        group/item relative flex h-11 w-full items-center transition-all duration-200 outline-none
        ${isActive ? "text-[#2e5c33]" : "text-[#7a8c80] hover:text-[#2e3c33]"}
      `}
    >
      {/* 激活背景悬浮于底层 */}
      <div className={`absolute inset-y-0 left-2 right-2 rounded-lg transition-colors 
        ${isActive ? "bg-[#e3f2e5]" : "group-hover/item:bg-[#f0f4f1]"}
      `} />
      
      {/* 图标容器严格 w-14 */}
      <span className="relative z-10 flex w-14 shrink-0 items-center justify-center">
        <Icon size={20} className="transition-transform group-hover/item:scale-110" />
      </span>

      {/* 文字标签绝对定位，不影响图标居中 */}
      <span className="absolute left-14 z-10 text-sm font-medium tracking-wide opacity-0 transition-opacity duration-300 group-hover:opacity-100">
        {label}
      </span>
    </button>
  );
};

// ==========================================
// 3. 主应用布局
// ==========================================
export default function App() {
  const [currentPath, setCurrentPath] = useState('/dashboard');
  const [isHovered, setIsHovered] = useState(false);
  const [isListOpen, setIsListOpen] = useState(false);
  const [selectedTranslator, setSelectedTranslator] = useState(TRANSLATORS[0]);

  // 侧边栏展开状态：鼠标悬停 或 列表打开 时保持展开
  const isExpanded = isHovered || isListOpen;

  const handleTranslatorSelect = (translator) => {
    // 如果点击的是自身（关闭），如果是新的则切换并关闭
    if (translator.id !== undefined) {
      setSelectedTranslator(translator);
    }
    setIsListOpen(false);
  };

  return (
    <div className="min-h-screen bg-[#f8fafc] flex font-sans antialiased">
      {/* 侧边栏 */}
      <nav 
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => {
          setIsHovered(false);
          setIsListOpen(false); // 鼠标彻底离开侧边栏区域（包含隐形桥梁）时才关闭
        }}
        className={`
          fixed left-0 top-0 z-50 flex h-screen flex-col bg-white border-r border-gray-100
          transition-[width] duration-400 ease-out group
          ${isExpanded ? 'w-56 shadow-xl' : 'w-14'}
        `}
      >
        {/* Logo (严格 w-14 对齐) */}
        <div className="relative flex h-16 w-full items-center mb-2">
          <div className="relative z-10 flex w-14 shrink-0 items-center justify-center">
            <Trees className="h-7 w-7 text-green-700" />
          </div>
          <span className="absolute left-14 text-lg font-black tracking-tighter text-gray-800 opacity-0 group-hover:opacity-100 transition-opacity duration-300 whitespace-nowrap">
            POPRKO <span className="text-green-600">HUB</span>
          </span>
        </div>

        {/* 异构汉化组按钮 (自带列表) */}
        <div className="mb-4">
          <TranslatorButton 
            activeTranslator={selectedTranslator}
            isOpen={isListOpen}
            onClick={(t) => t?.id ? handleTranslatorSelect(t) : setIsListOpen(!isListOpen)}
          />
        </div>

        {/* 主导航区域 */}
        <div className="flex-1 space-y-1">
          {MAIN_NAV.map((item) => (
            <SidebarItem
              key={item.id}
              icon={item.icon}
              label={item.label}
              isActive={currentPath === item.path}
              onClick={() => setCurrentPath(item.path)}
            />
          ))}
        </div>

        {/* 底部设置 */}
        <div className="py-2 border-t border-gray-50">
          <SidebarItem
            icon={FOOTER_NAV.icon}
            label={FOOTER_NAV.label}
            isActive={currentPath === FOOTER_NAV.path}
            onClick={() => setCurrentPath(FOOTER_NAV.path)}
          />
        </div>
      </nav>

      {/* 内容 */}
      <main className={`flex-1 ml-14 p-12 transition-all duration-400 ${isExpanded ? 'pl-20' : ''}`}>
        <div className="max-w-4xl">
          <h1 className="text-4xl font-black text-gray-900 tracking-tight leading-none mb-4">
            已修复：完美的异构悬浮对齐
          </h1>
          <p className="text-gray-500 text-lg mb-8">
            当前服务组：<span className="font-bold text-green-600">{selectedTranslator.name}</span>
          </p>
          
          <div className="bg-white p-6 rounded-2xl border border-gray-100 shadow-sm leading-relaxed text-gray-600 space-y-4">
            <p><strong className="text-green-600">1. 彻底解决收缩错误：</strong>我为弹出列表增加了一个 <code className="bg-gray-100 px-1 rounded">pl-3</code> 的“隐形桥梁”。当你把鼠标从深绿色按钮滑向右侧的列表时，你的鼠标实际上是在这个不可见的 Padding 上滑动。这保证了你的鼠标**永远没有离开 Sidebar 的有效判定区域**，侧边栏和列表将稳如泰山，绝对不会意外收起。</p>
            <p><strong className="text-green-600">2. 重构绝对居中：</strong>移除了导致错位的 <code className="bg-gray-100 px-1 rounded">px-1.5</code> 干扰。现在所有的组件：Logo、汉化组异构按钮、常规列表，其图标均被物理锁定在一个绝对的 <code className="bg-gray-100 px-1 rounded">w-14</code> 容器中。无论是否展开，中心点像素级对齐。</p>
            <p><strong className="text-green-600">3. 异构视觉更新：</strong>汉化组按钮现在变成了一个位于中心的正方形圆角卡片，展开后右侧才会浮现对应的文字区域，既与普通列表区分，又完美保持了轴线对齐。</p>
          </div>
        </div>
      </main>

      <style dangerouslySetInnerHTML={{__html: `
        @keyframes fade-in { from { opacity: 0; } to { opacity: 1; } }
        @keyframes slide-in-from-left-2 { from { transform: translateX(-8px); } to { transform: translateX(0); } }
        .animate-in { animation: fade-in 0.2s ease-out, slide-in-from-left-2 0.2s cubic-bezier(0.175, 0.885, 0.32, 1.275); }
      `}} />
    </div>
  );
}
