import React, { useState } from 'react';
import { Type, User, AlignLeft, Loader2, Layers, Hash } from 'lucide-react';
import type { CreateComicArgs } from "@/types";
import type { Result } from "@/types/utils/result";
import type { WorksetInfo } from "@/types/workset";

type Props = {
  currWorkset: WorksetInfo;
  onCreateComic: (args: CreateComicArgs) => Promise<Result<string>>;
  onClose: () => void;
};

/**
 * ComicCreatorModal - 极简明亮版 V2
 * 风格：纯白、淡绿触发、精简文案
 */
export default function ComicCreatorModal({ currWorkset, onCreateComic, onClose }: Props) {
  const [formData, setFormData] = useState({
    title: '',
    author: '',
    description: ''
  });
  const [isSubmitting, setIsSubmitting] = useState(false);

  const isValid = formData.title.trim() && formData.author.trim();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!isValid) return;

    setIsSubmitting(true);
    const result = await onCreateComic({
      teamId: currWorkset?.teamId || "default",
      ...formData
    });
    setIsSubmitting(false);

    if (result?.success) {
      onClose();
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-white/40 backdrop-blur-sm animate-in fade-in duration-200">
      <div className="w-full max-w-[280px] bg-white border border-slate-100 rounded-[24px] shadow-[0_8px_30px_rgb(0,0,0,0.04)] overflow-hidden animate-in zoom-in-95 duration-200">
        
        {/* Header */}
        <div className="pt-6 pb-2 text-center">
          <h3 className="text-[15px] font-bold text-slate-800">新建作品</h3>
          
          {/* Workset Info - 极简辅助标签 */}
          <div className="mt-3 flex items-center justify-center gap-1.5 px-4">
            <div className="flex items-center gap-1 px-2 py-0.5 bg-gray-50 rounded-md">
              <Layers className="w-2.5 h-2.5 text-slate-400" />
              <span className="text-[10px] text-slate-500 truncate max-w-[80px]">
                {currWorkset?.name || '默认'}
              </span>
            </div>
          </div>
        </div>

        <form onSubmit={handleSubmit} className="p-5">
          <div className="space-y-3">
            
            {/* Title */}
            <div className="flex items-center gap-2 px-3 py-2 bg-gray-50 rounded-xl focus-within:bg-white focus-within:ring-1 focus-within:ring-gray-200 transition-all">
              <Type className="w-3.5 h-3.5 text-slate-400" />
              <input
                type="text"
                required
                placeholder="标题"
                className="w-full bg-transparent text-sm text-slate-700 placeholder:text-slate-300 outline-none"
                value={formData.title}
                onChange={(e) => setFormData({ ...formData, title: e.target.value })}
              />
            </div>

            {/* Author */}
            <div className="flex items-center gap-2 px-3 py-2 bg-gray-50 rounded-xl focus-within:bg-white focus-within:ring-1 focus-within:ring-gray-200 transition-all">
              <User className="w-3.5 h-3.5 text-slate-400" />
              <input
                type="text"
                required
                placeholder="作者"
                className="w-full bg-transparent text-sm text-slate-700 placeholder:text-slate-300 outline-none"
                value={formData.author}
                onChange={(e) => setFormData({ ...formData, author: e.target.value })}
              />
            </div>

            {/* Description */}
            <div className="flex items-start gap-2 px-3 py-2 bg-gray-50 rounded-xl focus-within:bg-white focus-within:ring-1 focus-within:ring-gray-200 transition-all">
              <AlignLeft className="w-3.5 h-3.5 text-slate-400 mt-0.5" />
              <textarea
                placeholder="描述 (选填)"
                rows={2}
                className="w-full bg-transparent text-sm text-slate-700 placeholder:text-slate-300 outline-none resize-none"
                value={formData.description}
                onChange={(e) => setFormData({ ...formData, description: e.target.value })}
              />
            </div>
          </div>

          {/* Action Buttons - 并排设计 */}
          <div className="mt-6 flex items-center gap-2">
            <button
              type="button"
              onClick={onClose}
              className="flex-1 py-2 text-xs font-bold text-slate-400 bg-gray-50 hover:bg-gray-100 rounded-xl transition-all active:scale-95"
            >
              取消
            </button>
            <button
              type="submit"
              disabled={isSubmitting || !isValid}
              className={`
                flex-1 py-2 text-xs font-bold rounded-xl transition-all active:scale-95 flex items-center justify-center gap-1
                ${isValid 
                  ? 'bg-gray-50 text-slate-600 hover:bg-emerald-50 hover:text-emerald-600 border border-transparent hover:border-emerald-100' 
                  : 'bg-gray-50 text-slate-300 cursor-not-allowed'
                }
              `}
            >
              {isSubmitting ? (
                <Loader2 className="w-3 h-3 animate-spin" />
              ) : (
                "确认"
              )}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}
